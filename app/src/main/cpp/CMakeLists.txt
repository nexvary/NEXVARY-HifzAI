cmake_minimum_required(VERSION 3.22)
project(hifz_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# AGP Debug otherwise compiles CPU tensor kernels at -O0, making real inference unusable.
# Keep debug symbols and app debugging; optimize the native math in every variant.
add_compile_options(-O3)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_IS_DEV OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "No runtime networking" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "Portable Android binaries" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP_FETCH OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_RPC OFF CACHE BOOL "No remote backend" FORCE)
set(GGML_AVX OFF CACHE BOOL "" FORCE)
set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
set(GGML_FMA OFF CACHE BOOL "" FORCE)
set(GGML_F16C OFF CACHE BOOL "" FORCE)
include(FetchContent)
FetchContent_Declare(whisper
    URL https://github.com/ggml-org/whisper.cpp/archive/d1be6fde11ac6e0407606b4e42fe72d34add8037.tar.gz
    URL_HASH SHA256=645445da884df6dbd52b30f82d7be7c01d1767c938714e477e786b88d529e9f4)
FetchContent_MakeAvailable(whisper)
add_library(hifz_whisper SHARED hifz_whisper.cpp)
target_include_directories(hifz_whisper PRIVATE ${whisper_SOURCE_DIR}/include)
target_link_libraries(hifz_whisper PRIVATE whisper)
# Keep a consistent 16KiB ELF layout, including NDK 32-bit outputs (4KiB OS pages still work).
target_link_options(hifz_whisper PRIVATE "-Wl,-z,max-page-size=16384")
